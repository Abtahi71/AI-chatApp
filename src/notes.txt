type User{
    id:ID!
    name:String!
    email:String!
}

type Query{
    users:[User]
    user(id:ID!):User
}

const resolver=()=>{
    Query{
        users:()=>users,
        user:({id})=>users.find(u=>u.id===id)
    }
}
